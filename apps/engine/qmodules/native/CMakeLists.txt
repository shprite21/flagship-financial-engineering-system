cmake_minimum_required(VERSION 3.24)
project(ParallelABM VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ABM_CUDA "Build the CUDA backend (requires NVIDIA Toolkit)" OFF)
add_library(abm simulation/cpu/economy.cpp)
target_include_directories(abm PUBLIC simulation)
if(ABM_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(abm PRIVATE simulation/cuda/economy.cu)
  target_compile_definitions(abm PUBLIC ABM_HAS_CUDA=1)
  set_target_properties(abm PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(abm PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  target_link_libraries(abm PUBLIC CUDA::cudart)
endif()
if(MSVC)
  target_compile_options(abm PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/fp:strict>)
else()
  target_compile_options(abm PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-ffp-contract=off>")
endif()
add_executable(abm_cli simulation/experiments/main.cpp)
target_link_libraries(abm_cli PRIVATE abm)
enable_testing()
add_executable(abm_tests tests/test_engine.cpp)
target_link_libraries(abm_tests PRIVATE abm)
add_test(NAME engine COMMAND abm_tests)
