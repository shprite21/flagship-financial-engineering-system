cmake_minimum_required(VERSION 3.24)
project(MonteCarloCUDA VERSION 1.0 LANGUAGES CXX)
option(MC_ENABLE_CUDA "Build the NVIDIA CUDA backend" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mc_core src/pricing.cpp)
target_include_directories(mc_core PUBLIC include)
if(MSVC)
  target_compile_options(mc_core PRIVATE /W4)
else()
  target_compile_options(mc_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MC_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(mc_gpu src/pricing_cuda.cu)
  target_include_directories(mc_gpu PUBLIC include)
  target_link_libraries(mc_gpu PUBLIC mc_core CUDA::cudart CUDA::curand)
  set_target_properties(mc_gpu PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
else()
  add_library(mc_gpu src/pricing_stub.cpp)
  target_link_libraries(mc_gpu PUBLIC mc_core)
endif()
add_executable(mc_benchmark src/main.cpp)
target_link_libraries(mc_benchmark PRIVATE mc_core mc_gpu)
include(CTest)
if(BUILD_TESTING)
  add_executable(mc_tests tests/test_pricing.cpp)
  target_link_libraries(mc_tests PRIVATE mc_core mc_gpu)
  add_test(NAME pricing_cpu COMMAND mc_tests)
  if(MC_ENABLE_CUDA)
    add_test(NAME pricing_gpu COMMAND mc_tests --gpu)
    add_executable(mc_rng_tests tests/test_rng.cu)
    target_link_libraries(mc_rng_tests PRIVATE CUDA::cudart CUDA::curand)
    set_target_properties(mc_rng_tests PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    add_test(NAME rng_cpu_gpu COMMAND mc_rng_tests)
    add_test(NAME gpu_unavailable_error COMMAND ${CMAKE_COMMAND}
      -DBENCHMARK=$<TARGET_FILE:mc_benchmark> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/expect_no_gpu.cmake)
    set_tests_properties(gpu_unavailable_error PROPERTIES ENVIRONMENT "CUDA_VISIBLE_DEVICES=-1")
  endif()
  add_test(NAME cli_help COMMAND mc_benchmark --help)
  add_test(NAME cli_invalid_samples COMMAND mc_benchmark --samples 1 --backend cpu)
  set_tests_properties(cli_invalid_samples PROPERTIES WILL_FAIL TRUE)
endif()
